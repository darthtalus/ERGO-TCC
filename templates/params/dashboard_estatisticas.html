{# Methodology / Technical Logic #}
{% trans "Methodology & Calculation Logic" %}

1. {% trans "Daily Effectiveness Records" %}

{% trans "Source Table:" %} daily_effectiveness
{% trans "When a user clicks 'Finish the Day', a record is created with:" %}
• user: {% trans "the user who finished the day" %}
• day: {% trans "current date (auto-generated)" %}
• effectiveness: {% trans "score from 1 to 5 chosen by the user" %}
{% trans "Constraint: Only one record per user per day (unique_together: user, day)." %}

2. {% trans "Aspect History Snapshot" %}

{% trans "Source Table:" %} measurement_history
{% trans "At the same time as the daily record, ALL parameters configured in aspect_measurement are copied to this table, preserving:" %}
• dimension: {% trans "ENVIRONMENT, COGNITION, or BEHAVIOR" %}
• nome: {% trans "aspect name (e.g., Temperature, Focus)" %}
• valor: {% trans "absolute value (e.g., 24°C, High)" %}
• parametrized: {% trans "standardized score from 1 to 5" %}
• effectiveness_id: {% trans "link to the daily record" %}

3. {% trans "Task Evaluations" %}

{% trans "Source Table:" %} task_evaluation
{% trans "When a task status changes to 'Done', evaluations are recorded:" %}
• rate_type = SELF: {% trans "the executor evaluates their own effectiveness (1-5)" %}
• rate_type = STAKEHOLDER: {% trans "stakeholders evaluate the executor's effectiveness (1-5)" %}
• environment, cognition, behavior: {% trans "dimension scores assigned during evaluation" %}

4. {% trans "ABC Dimensions Impact" %}

{% trans "Source Table:" %} measurement_history
{% trans "The dimension averages shown in the graph and summary table are calculated exclusively from the parametrized field (1-5 scale) of the daily aspect snapshots." %}
{% trans "Formula:" %} AVG(parametrized) WHERE dimension = 'X'
{% trans "Important: These values reflect the executors' daily self-assessments of their working conditions, NOT the task-level evaluations. Stakeholders do not evaluate these dimensions." %}

{% trans "The Evolution Analysis graph correlates both data sources by matching effectiveness.day (daily record date) with DATE(taskrate.created_at) (task evaluation date). This allows identifying discrepancies between the user's general perception and their actual task performance." %}
